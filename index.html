// --- CANVAS COLOURING ENGINE --- const canvas = document.getElementById('paintCanvas'); const ctx = canvas.getContext('2d'); let painting = false; function initCanvas() { // Set canvas resolution size inside wrapper canvas.width = 400; canvas.height = 400; clearCanvas(); // Event Listeners for Touch & Mouse canvas.addEventListener('mousedown', startPosition); canvas.addEventListener('mouseup', endPosition); canvas.addEventListener('mouseleave', endPosition); // Tambah ini supaya berhenti melukis jika kursor keluar canvas canvas.addEventListener('mousemove', draw); canvas.addEventListener('touchstart', (e) => { startPosition(e.touches[0]); }, { passive: false }); canvas.addEventListener('touchend', endPosition); canvas.addEventListener('touchmove', (e) => { draw(e.touches[0]); }, { passive: false }); } function startPosition(e) { painting = true; ctx.beginPath(); // Mulakan garisan baru draw(e); } function endPosition() { painting = false; ctx.beginPath(); // Putuskan garisan lama } function draw(e) { if (!painting) return; // Dapatkan kedudukan tepat tetikus/sentuhan pada canvas const rect = canvas.getBoundingClientRect(); const x = (e.clientX - rect.left) * (canvas.width / rect.width); const y = (e.clientY - rect.top) * (canvas.height / rect.height); ctx.lineWidth = 12; // Ketebalan berus warna sesuai untuk kanak-kanak ctx.lineCap = 'round'; // Hujung berus bulat ctx.lineJoin = 'round'; ctx.strokeStyle = selectedPaintColor; ctx.lineTo(x, y); ctx.stroke(); ctx.beginPath(); ctx.moveTo(x, y); } function setPaintColor(color) { selectedPaintColor = color; // Kemaskini visual butang warna yang aktif document.querySelectorAll('.color-palette button').forEach(btn => { btn.classList.remove('scale-110', 'ring-4', 'ring-purple-400'); }); // Tambah highlight pada butang yang dipilih event.target.classList.add('scale-110', 'ring-4', 'ring-purple-400'); } function clearCanvas() { ctx.fillStyle = '#FFFFFF'; ctx.fillRect(0, 0, canvas.width, canvas.height); // Lukis outline ringkas (contoh: bulatan tengah) supaya kanak-kanak boleh mewarna di dalamnya ctx.beginPath(); ctx.arc(200, 200, 100, 0, 2 * Math.PI); ctx.lineWidth = 5; ctx.strokeStyle = '#cbd5e1'; ctx.stroke(); } // Fungsi log ke Firebase (ditambah perlindungan sekiranya firebase lambat/gagal) function logToFirebase(moduleName) { if (typeof database !== 'undefined') { try { database.ref('analytics/clicks').push({ module: moduleName, timestamp: Date.now(), language: currentLang }); } catch (err) { console.warn("Gagal menghantar data ke Firebase:", err); } } }
